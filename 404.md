<!--
  Referral links arrive here when they use the pretty `/r/CODE` form.

  GitHub Pages is static and has no rewrite rules, so `/r/ARC7K9QX2` is an unmatched path and
  gets served this page. Rather than showing someone a 404 for a link a friend sent them, hand
  the code to the real referral page.

  Shared links use `/r/?c=CODE` precisely so they do not land here: that form is a genuine 200
  with working Open Graph tags, which matters because these links are pasted into chat apps
  whose scrapers ignore a 404 response. This block is the safety net for a link that was
  shortened, retyped, or written by hand.
-->
<script>
(function () {
  var segments = window.location.pathname.split('/').filter(Boolean);
  if (segments.length < 2 || segments[segments.length - 2] !== 'r') return;
  var code = segments[segments.length - 1];
  if (!/^[A-Za-z0-9]{1,16}$/.test(code)) return;
  window.location.replace('/r/?c=' + encodeURIComponent(code.toUpperCase()));
})();
</script>
