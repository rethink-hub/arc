<!--
  Referral and route links arrive here when they use the pretty path form.

  GitHub Pages is static and has no rewrite rules, so `/r/ARC7K9QX2` and `/go/settings` are both
  unmatched paths and get served this page. Rather than showing someone a 404 for a link a friend
  sent them, hand the code or route to the page that knows what to do with it.

  Shared links use the query form — `/r/?c=CODE`, `/go/?to=ROUTE` and `/s/?c=SLUG` — precisely so they do not
  land here: those are genuine 200s with working Open Graph tags, which matters because these
  links are pasted into chat apps whose scrapers ignore a 404 response. This block is the safety
  net for a link that was shortened, retyped, or written by hand.
-->
<script>
(function () {
  var segments = window.location.pathname.split('/').filter(Boolean);
  if (segments.length < 2) return;

  if (segments[segments.length - 2] === 'r') {
    var code = segments[segments.length - 1];
    if (!/^[A-Za-z0-9]{1,16}$/.test(code)) return;
    window.location.replace('/r/?c=' + encodeURIComponent(code.toUpperCase()));
    return;
  }

  // `/s/SLUG` — a marketing link written in the pretty form. The slug is not validated here;
  // /s/ matches it against the real campaign table and falls back to an untagged store link.
  if (segments[0] === 's') {
    var slug = segments[1];
    if (!/^[a-z0-9._-]{1,64}$/i.test(slug)) return;
    window.location.replace('/s/?c=' + encodeURIComponent(slug.toLowerCase()));
    return;
  }

  // `/go/ROUTE`, including the one two-segment route (`/go/custom-actions/new`). The route is
  // not validated here — /go/ matches it against the real list and falls back to home — but the
  // shape is, so a stray path cannot be bounced onward as if it were a destination.
  if (segments[0] === 'go') {
    var route = segments.slice(1).join('/');
    if (!/^[a-z0-9_-]+(\/[a-z0-9_-]+)?$/i.test(route)) return;
    window.location.replace('/go/?to=' + encodeURIComponent(route.toLowerCase()));
  }
})();
</script>
